1. table barang
field: id, kode_barang(varchar), nama(varchar), kategori(varchar), stok_saat ini(int), stok_minimum(int), tipe_barang(varchar), garansi_bulan(int), ukuran(varchar)
2. table buyers
field: id, nama(varchar), kontak(varchar), alamat(text)
3. table suppliers
field: id, nama(varchar), kontak(varchar), alamat(text)
4. table stok_transactions
field: id, barang_id(int), tipe_transaksi(varchar), jumlah(int), tanggal(timestamp), keterangan(text), supplier_id(int), buyer_id(int)
5. table users
field: id, email(varchar), password(varchar), nama(varchar), role(varchar)
